\begin{recipe}[
	preparationtime={\unit[20]{min}},
	bakingtime={\unit[60]{min}},
	portion=\portion{12-16},
	bakingtemperature={\protect\bakingtemperature{fanoven=\unit[350]{\faren}}},
	source=\href{https://food52.com/recipes/86166-corn-mochi-cake-recipe}{Food52}
	]{Cheesy Corn Mochi Cake}\index{Cake}\index{Corn}\index{Vegetarian}\index{Kosher!Passover}
\ingredients[9]{%
	\unit[\nicefrac{1}{4}]{cup} & unsalted melted butter \\
	\unit[80]{g} & grated Pecorino\\
	\unit[1]{tsp} & chile flakes \\
	\unit[4-5]{cobs} & corn \\
	\unit[170]{g} & sour cream \\
	2 & large eggs \\
	\unit[124]{g} & sugar \\
	\unit[1\nicefrac{3}{4}]{tsp} & salt \\
	\unit[254]{g} & mochiko (glutinous sweet rice flour) \\
}
\preparation{
	\step Shuck the corn and remove the silk. Using a knife, cut the kernels from the cobs (use a large bowl with a small upside-down bowl inside to aid with the process). Use the back of the knife to scrape the cobs and get the corn milk. You should have about \unit[2]{cups} of corn kernels, but a little more is fine.
	\step Preheat the oven to \unit[350]{\faren}. Butter a 8-inch square baking pan or 9-inch round cake pan.
	\step In a small bowl, toss the grated Pecorino with the chile flakes. Set aside \unit[\nicefrac{1}{2}]{cup} of the cheese mixture for later. Coat the bottom and sides of the prepared pan with the remaining cheese mixture.
	\step In a blender, combine \unit[1]{cup} of corn kernels with the sour cream, and \nicefrac{1}{2} of the reserved cheese mixture. Blend until combined.
	\step In a medium bowl, whisk together the eggs and melted butter until smooth. Add the sugar, salt, and corn puree and whisk until combined. Add the mochiko and whisk until combined. Add the remaining corn kernels and stir to combine.
	\step Move the batter to the prepared pan and sprinkle the remaining cheese mixture on top. Bake until deeply browned and the cake bounces back in the center, about 55-60 minutes. Cool at least 15 minutes before inverting and serving.
}
